FROM python:3.11-slim

# System deps (libzbar for pyzbar, etc.)
RUN apt-get update && apt-get install -y --no-install-recommends \
    libzbar0 \
    libzbar-dev \
    libjpeg62-turbo \
    zlib1g \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY bot.py .

# Render PORT env var use करेगा, fallback 8080
ENV PORT=8080

CMD ["python", "-u", "bot.py"]
